# Integration test cases: one shell command per line.
# Empty lines and lines starting with `#` are skipped.
# Each line is run as:  ./42sh -c "<line>"   vs   bash --posix -c "<line>"
# stdout, stderr, and exit status must match. Each line is also re-run
# under valgrind against ./42sh only (skipped with VALGRIND=0).

# --- basic tests ---------------------------------------------------------
ls | sort -rn | cat -e
ls > /tmp/ftsh_ls_out /
cat /tmp/ftsh_ls_out
< /tmp/ftsh_ls_out cat -e >> /tmp/ftsh_ls_out2
echo 1 >out >&2 2>err
printf 'a\nb\nc\n' | base64 | base64 -d | grep -c b | sed -e 's/1/Yes/g' -e 's/0/No/g'
ps a | grep 'base64' | grep -v 'grep'
touch test_file; ls -1 test_file
exit 1 | exit 2 | exit 3; echo "stayin' alive"
echo out >&-; echo out2
echo out >&- | echo out2
echo out >&- && echo out2
echo out >&- || echo out2


# --- echo / basic words --------------------------------------------------
echo hello
echo hello world
echo
echo a b   c

# --- exit status ---------------------------------------------------------
true
false
exit 0
exit 7
exit 42

# --- sequential lists ----------------------------------------------------
echo a; echo b
true; false; true
false; echo after

# --- and / or chains -----------------------------------------------------
true && echo yes
false && echo nope
false || echo no
true || echo nope
true && echo a && echo b
false || echo a || echo b

# --- pipelines -----------------------------------------------------------
echo hi | cat
echo one | cat | cat
printf '%s\n' a b c | cat
echo abc | tr a-z A-Z

# --- subshells / groups --------------------------------------------------
(echo inside)
(true && echo ok)
(false); echo $?
